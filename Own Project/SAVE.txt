Lenght:
string mcx = "HakimMCX";

Indexs:
string mcx = "HakimMCX";
cout << mcx[0]; //M
cout << mcx[mcx.length() - 1]; //X
mcx[0] = "J";
cout << mcx; //JakimMCX

At:
string mcx = "HakimMCX";
cout << mcx.at(0);  //M
cout << mcx.at(mcx.length() - 1) //X
mcx.at(0) = "J"
cout << mcx; //JakimMCX

Boolean + If-else:

int votingAge = 18;
if (myAge >= votingAge) {
  cout << "Old enough to vote!";
} else {
  cout << "Not old enough to vote.";
}

Ternary Operator (Ringkasan if-else):
variable = (condition) ? expressionTrue : expressionFalse;

int time = 20;
string result = (time < 18) ? "Good day." : "Good evening.";
cout << result;

or

int time = 20;
cout << ((time < 18) ? "Good day." : "Good evening."); //print terus


Nested Ternary:
int time = 22;
string message = (time < 12) ? "Good morning."
  : (time < 18) ? "Good afternoon."
  : "Good evening.";
cout << message;

Nested Loops:
for(int i=1;i<=2;++i){
	cout << i << endl; // Executes 2 times
	for(int j=1;j<=3;++j){
		cout << j << endl;  // Executes 6 times (2 * 3)
	}
}

Foreach Loop:
int myNumbers[5] = {10, 20, 30, 40, 50};
for (int num : myNumbers) {
  cout << num << "\n";
}

string word = "Hello";
for (char c : word) {
  cout << c << "\n";
}
